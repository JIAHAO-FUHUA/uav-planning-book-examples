cmake_minimum_required(VERSION 3.16)
project(UAVPlanningBookExamples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()

set(GRID_DIR "${CMAKE_CURRENT_SOURCE_DIR}/examples/chapter04/Chapter_4.2_Code")
set(TRAJECTORY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/examples/chapter04/Chapter_4.3_Code")
add_executable(grid_planning "${GRID_DIR}/planning.cpp")
add_executable(sampling_dynamics "${GRID_DIR}/sampling_dynamics.cpp")
add_executable(simulations "${GRID_DIR}/simulations.cpp")
add_executable(trajectory_generation "${TRAJECTORY_DIR}/trajectory_generation.cpp")

foreach(target grid_planning sampling_dynamics simulations trajectory_generation)
  # Keep assert-based self-tests active in Release builds as well.
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ${target}_self_test COMMAND ${target} --self-test)
  set_tests_properties(${target}_self_test PROPERTIES TIMEOUT 180)
endforeach()
